import java.util.Scanner;
class TelephoneBill {
// Member variables
int customerNumber;
String customerName;
int prevMonthDuration;
int currMonthDuration;
String connectionType; // "prepaid" or "postpaid"
double billAmount;
// Method to input data
void inputData() {
Scanner sc = new Scanner(System.in);
System.out.print("Enter Customer Number: ");
customerNumber = sc.nextInt();
sc.nextLine(); // consume newline
System.out.print("Enter Customer Name: ");
customerName = sc.nextLine();
System.out.print("Enter Previous Month Call Duration (in minutes): ");
prevMonthDuration = sc.nextInt();
System.out.print("Enter Current Month Call Duration (in minutes): ");
currMonthDuration = sc.nextInt();
sc.nextLine(); // consume newline
System.out.print("Enter Type of Connection (prepaid/postpaid): ");
connectionType = sc.nextLine().toLowerCase();
}
// Method to calculate bill
void calculateBill() {
int minutes = currMonthDuration;
billAmount = 0;
if (connectionType.equals("prepaid")) {
if (minutes <= 100) {
billAmount = minutes * 1.0;
} else if (minutes <= 200) {
billAmount = 100 * 1.0 + (minutes - 100) * 1.5;
} else {
billAmount = 100 * 1.0 + 100 * 1.5 + (minutes - 200) * 2.0; }
} else if (connectionType.equals("postpaid")) {
if (minutes <= 100) {
billAmount = minutes * 0.75;
} else if (minutes <= 200) {
billAmount = 100 * 0.75 + (minutes - 100) * 1.25;
} else {
billAmount = 100 * 0.75 + 100 * 1.25 + (minutes - 200) * 1.75;
}
} else {
System.out.println("Invalid connection type! Enter prepaid or postpaid.");
}
}
// Method to display bill
void displayBill() {
System.out.println("\n------ TELEPHONE BILL ------");
System.out.println("Customer Number : " + customerNumber);
System.out.println("Customer Name : " + customerName);
System.out.println("Previous Duration : " + prevMonthDuration + " minutes");
System.out.println("Current Duration : " + currMonthDuration + " minutes");
System.out.println("Connection Type : " + connectionType);
System.out.printf("Total Bill Amount : Rs. %.2f\n", billAmount);
System.out.println("----------------------------");
}
}
public class Main {
public static void main(String[] args) {
TelephoneBill tb = new TelephoneBill(); // create object
tb.inputData(); // take input
tb.calculateBill(); // calculate
tb.displayBill(); // display
}
}
